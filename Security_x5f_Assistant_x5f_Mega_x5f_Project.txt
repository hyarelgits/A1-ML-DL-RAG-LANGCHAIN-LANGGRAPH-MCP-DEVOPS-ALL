# IMPORTS - Python libraries
import re  # For finding words - Used in RAG

# ---------- TOKEN - OPTIMIZATION: COST CONTROL ----------
# This is for OPTIMIZATION - To count tokens and save cost
# If tokens are high, we know we are wasting money
def count_tokens(text):
    # 1 token = about 4 chars - Simple rule for learning
    return len(text) // 4 + 1

# ---------- NEURAL NETWORK - BRAIN ----------
# This is for UNDERSTANDING - How AI brain works
# Input -> Weight -> Output
def simple_neuron(question):
    # This is for WEIGHT - Security word has high weight 0.9
    score = 0
    if "security" in question.lower():
        score += 0.9  # High weight = Important
    if "crm" in question.lower():
        score += 0.2  # Low weight = Less important for security
    if "lms" in question.lower():
        score += 0.2
    return score

# ---------- MoE ROUTER - OPTIMIZATION: MEMORY SAVING ----------
# This is for OPTIMIZATION - Memory saving 97 percent
# Only 1 expert works, others sleep. Memory use 1-3 percent only.
def router(question):
    # This is for ROUTING - Decide which expert to call
    q = question.lower()
    if "security" in q or "incident" in q or "p1" in q or "soc" in q:
        return "security"  # Call Security Expert only
    if "crm" in q or "customer" in q:
        return "crm"  # Call CRM Expert only
    if "lms" in q or "student" in q:
        return "lms"  # Call LMS Expert only
    return "security"  # Default

def load_expert_doc(expert):
    # This is for LOADING DOC - Load only 1 expert doc, not all docs
    # This is for OPTIMIZATION - Saves memory and tokens
    if expert == "security":
        path = "docs/security_sop.txt"
    elif expert == "crm":
        path = "docs/crm_sop.txt"
    else:
        path = "docs/lms_sop.txt"
    with open(path, encoding="utf-8") as f:
        return f.read()

# ---------- RAG - OPTIMIZATION: TOKEN SAVING 99.7 PERCENT ----------
# This is for OPTIMIZATION - Token saving
# Instead of sending all 500 docs = 500,000 tokens, send only 1 chunk = 1,500 tokens
def rag_retrieve(question, doc):
    # This is for RETRIEVAL - Find best line only
    lines = [l for l in doc.split("\n") if l.strip()]
    q_words = set(re.findall(r"\w+", question.lower()))
    # Find line with maximum matching words
    best = max(lines, key=lambda line: len(set(re.findall(r"\w+", line.lower())) & q_words))
    return best

# ---------- LoRA - OPTIMIZATION: TRAINING COST SAVING ----------
# This is for OPTIMIZATION - Training cost saving
# Full fine-tuning costs Rs 15,000. LoRA adapter costs Rs 500. Train once, use many times.
# LoRA teaches HOW to answer - Format, slang, tool calling. Facts come from RAG.
COMPANY_FORMAT = "Incident ID: {id} Severity: {severity} Summary: {summary} Steps: {steps} Owner: SOC Team"
SLANG = {"p1": "Critical - Fix in 1 hour", "soc": "Security Operations Center"}

def apply_lora_format(question, rag_chunk):
    # This is for FORMAT - Company format never changes, so train once
    severity = "Critical" if "p1" in question.lower() else "Medium"
    summary = rag_chunk
    # This is for SLANG - Replace P1 with full meaning
    for k, v in SLANG.items():
        summary = summary.replace(k.upper(), v)
    return COMPANY_FORMAT.format(id="INC-101", severity=severity, summary=summary, steps="1. Create ticket 2. Inform SOC 3. Isolate server")

# ---------- QUANTIZATION - OPTIMIZATION: MEMORY SAVING 75 PERCENT ----------
# This is for OPTIMIZATION - Memory saving
# FP32 28 GB -> INT8 7 GB -> INT4 4 GB. Same knowledge, smaller memory, faster, cheaper.
def quantize_example():
    weight_fp32 = 3.1415926535  # 32-bit, big, bulky, slow
    weight_int8 = round(weight_fp32, 1)  # 8-bit idea, small, fast, cheap - 3.1
    return f"FP32: {weight_fp32} (28GB) -> INT8: {weight_int8} (7GB) - 75 percent smaller, 97 percent accuracy"

# ---------- HALLUCINATION GUARD - OPTIMIZATION: SAFETY ----------
# This is for SAFETY - Stop AI from lying
# If answer has fake phone, block it. If answer not in docs, say I do not know.
def hallucination_guard(answer, rag_chunk):
    # This is for GUARD - Check if phone number is invented
    if "1800" in answer and "security@company.com" not in rag_chunk:
        return answer + "\n[GUARD]: No phone in docs. Contact is security@company.com only. Do not invent."
    if "phone" in answer.lower() and "phone" not in rag_chunk.lower():
        return "I do not know - Not in documents. Please check with admin. I will not make up answer."
    return answer

# ---------- TTS - OPTIMIZATION: USER EXPERIENCE ----------
# This is for USER EXPERIENCE - Voice helps drivers, visually impaired
# Real project uses Azure TTS / ElevenLabs. Here we simulate with print.
def tts_speak(text):
    return f"[TTS] Speaking: {text[:80]}... (In real project, Azure TTS will speak this)"

# ---------- MAIN - COMBINE ALL - FULL WORKFLOW ----------
# This is for WORKFLOW - Combine all tools in order
def ask(question):
    print(f"\n=== Question: {question} ===")
    print(f"Tokens in question: {count_tokens(question)}")  # Token counting
    print(f"Neural score: {simple_neuron(question):.2f}")  # Brain score
    expert = router(question)  # MoE Router
    print(f"MoE Router -> Expert: {expert} (Other experts sleeping - Saves cost)")
    doc = load_expert_doc(expert)  # Load only 1 doc
    print(f"Total doc tokens if we send all: {count_tokens(doc)}")
    chunk = rag_retrieve(question, doc)  # RAG - Only 1 chunk
    print(f"RAG retrieved only 1 chunk: '{chunk[:80]}...' Tokens: {count_tokens(chunk)} - 99 percent tokens saved")
    formatted = apply_lora_format(question, chunk)  # LoRA - Format
    print(f"LoRA Adapter applied -> Company Format + Slang P1/SOC")
    guarded = hallucination_guard(formatted, chunk)  # Guard - Safety
    print(f"\n--- Final Answer ---\n{guarded}")
    print(f"\n{tts_speak(guarded)}")  # TTS - Voice
    print(f"\nQuantization: {quantize_example()}")  # Quantization - Memory
    return guarded

# ---------- RUN 3 TEST QUESTIONS ----------
if __name__ == "__main__":
    ask("What is P1 security incident process?")
    ask("How to add student in LMS?")
    ask("What is security incident phone number?")
